BEGINNER1. Why Generics Exist

Definition

Generics let you write functions, interfaces, and classes that work with any type while still preserving type information, instead of falling back to any and losing type safety.

Real-world analogy: A generic function is like a shipping container — it doesn't care what's inside, but it guarantees whatever comes out is exactly what went in, whether that's books, furniture, or electronics.

The Problem Without Generics

// Using 'any' loses all type information
function firstElementAny(arr: any[]): any {
    return arr[0];
}

const num = firstElementAny([1, 2, 3]);
num.toUpperCase();   // ❌ Compiles fine, but crashes at runtime! No safety.

The Fix: A Generic Function

function firstElement(arr: T[]): T {
    return arr[0];
}

const num = firstElement([1, 2, 3]);        // T inferred as number
const str = firstElement(["a", "b", "c"]);  // T inferred as string

// num.toUpperCase();   // ❌ Error: Property 'toUpperCase' does not exist
//                          on type 'number' - caught at compile time!

BEGINNER2. Generic Functions

function identity(value: T): T {
    return value;
}

identity("hello");   // explicit type argument
identity(42);                 // inferred as number - most common usage

Generic Functions Over Arrays

function last(arr: T[]): T | undefined {
    return arr[arr.length - 1];
}

last([1, 2, 3]);            // 3 (number | undefined)
last(["x", "y", "z"]);      // "z" (string | undefined)
last([]);                    // undefined

Generic Arrow Functions

const wrapInArray = (value: T): T[] => [value];
// Note: the trailing comma in  avoids ambiguity with JSX in .tsx files

wrapInArray(5);        // number[]
wrapInArray("hi");     // string[]

INTERMEDIATE3. Generic Interfaces and Type Aliases

interface Box {
    value: T;
}

const numberBox: Box = { value: 42 };
const stringBox: Box = { value: "hello" };

interface ApiResponse {
    data: T;
    status: number;
    error?: string;
}

const response: ApiResponse<{ id: number; name: string }> = {
    data: { id: 1, name: "Widget" },
    status: 200
};

Generic Type Aliases

type Pair = { first: A; second: B };

const coordinate: Pair = { first: 10, second: 20 };
const entry: Pair = { first: "age", second: 30 };

type Nullable = T | null;

let maybeName: Nullable = null;
maybeName = "Reese";

INTERMEDIATE4. Generic Classes

class Stack {
    private items: T[] = [];

    push(item: T): void {
        this.items.push(item);
    }

    pop(): T | undefined {
        return this.items.pop();
    }

    peek(): T | undefined {
        return this.items[this.items.length - 1];
    }

    get size(): number {
        return this.items.length;
    }
}

const numberStack = new Stack();
numberStack.push(1);
numberStack.push(2);
console.log(numberStack.pop());   // 2

const stringStack = new Stack();
stringStack.push("a");
stringStack.push("b");

Generic Class with a Constrained Method

class Repository {
    private items: T[] = [];

    add(item: T): void {
        this.items.push(item);
    }

    findById(id: number): T | undefined {
        return this.items.find((item) => item.id === id);
    }
}

interface Product { id: number; name: string; }
const products = new Repository();
products.add({ id: 1, name: "Widget" });

INTERMEDIATE5. Constraints with extends

A generic constraint restricts T to types that have certain members, so you can safely use those members inside the function body.

interface HasLength {
    length: number;
}

function logLength(value: T): T {
    console.log(value.length);   // ✅ OK - guaranteed by the constraint
    return value;
}

logLength("hello");        // 5 - strings have .length
logLength([1, 2, 3]);      // 3 - arrays have .length
// logLength(42);           // ❌ Error: number doesn't have .length

Constraining to Keys of Another Type

function pluck(obj: T, key: K): T[K] {
    return obj[key];
}

const user = { id: 1, name: "Wren", active: true };

pluck(user, "name");     // "Wren" (string)
pluck(user, "active");   // true (boolean)
// pluck(user, "email");  // ❌ Error: 'email' is not a key of 'user'

INTERMEDIATE6. Default Type Parameters

interface ApiResult {
    data: T;
    success: boolean;
}

// Uses the default 'unknown' when no type argument is given
const genericResult: ApiResult = { data: "anything", success: true };

// Explicit type argument overrides the default
const typedResult: ApiResult = { data: [1, 2, 3], success: true };

Defaults Referencing Other Parameters

interface Cache {
    get(key: K): V | undefined;
    set(key: K, value: V): void;
}

// Cache === Cache since V defaults to K

ADVANCED7. Multiple Type Parameters

function merge(a: T, b: U): T & U {
    return { ...a, ...b };
}

const merged = merge({ name: "Ash" }, { age: 25 });
// merged: { name: string } & { age: number }
console.log(merged.name, merged.age);   // Ash 25

Mapping Between Two Related Generic Parameters

function zip(first: T[], second: U[]): [T, U][] {
    const length = Math.min(first.length, second.length);
    const result: [T, U][] = [];
    for (let i = 0; i < length; i++) {
        result.push([first[i], second[i]]);
    }
    return result;
}

zip([1, 2, 3], ["a", "b", "c"]);
// [[1, "a"], [2, "b"], [3, "c"]]

ADVANCED8. keyof and Generic Property Access

keyof produces a union of a type's property names as string literal types, commonly paired with generics for type-safe property access.

interface Settings {
    theme: string;
    fontSize: number;
    notifications: boolean;
}

type SettingsKey = keyof Settings;   // "theme" | "fontSize" | "notifications"

function updateSetting(
    settings: Settings,
    key: K,
    value: Settings[K]
): Settings {
    return { ...settings, [key]: value };
}

const settings: Settings = { theme: "dark", fontSize: 14, notifications: true };
updateSetting(settings, "fontSize", 16);      // ✅ OK - number matches
// updateSetting(settings, "fontSize", "16"); // ❌ Error: string not assignable to number
💡 Tip: Settings[K] is an "indexed access type" — it looks up the exact type of the property named by K, keeping the value parameter's type locked to whichever key was passed.

ADVANCED9. Generic Utility Type Patterns

Many of TypeScript's built-in utility types (fully explored in the Advanced & Utility Types guide) are themselves generics you can learn from and mimic.

// A simplified reimplementation of Partial
type MyPartial = {
    [K in keyof T]?: T[K];
};

interface Task { id: number; title: string; done: boolean; }

type PartialTask = MyPartial;
// { id?: number; title?: string; done?: boolean; }

function updateTask(task: Task, changes: MyPartial): Task {
    return { ...task, ...changes };
}

A Generic Result Wrapper

type Result =
    | { ok: true; value: T }
    | { ok: false; error: E };

function tryParseJson(json: string): Result {
    try {
        return { ok: true, value: JSON.parse(json) as T };
    } catch (e) {
        return { ok: false, error: e as Error };
    }
}

const result = tryParseJson<{ id: number }>('{"id": 1}');
if (result.ok) {
    console.log(result.value.id);   // ✅ narrowed - value exists
}

ADVANCED10. Best Practices

1. Only Add a Type Parameter When It's Actually Used Twice

// ❌ Unnecessary - T only appears once, 'unknown' would do
function log(value: T): void {
    console.log(value);
}

// ✅ Good - T links input and output, must be generic
function identity(value: T): T {
    return value;
}

2. Constrain Generics to the Minimum Shape You Need

// ✅ Good - only requires 'length', works with strings, arrays, etc.
function isEmpty(value: T): boolean {
    return value.length === 0;
}

3. Use Descriptive Names for Complex Generics

// ✅ Good for multi-parameter generics
interface Cache { /* ... */ }

// T, U, K, V are fine for simple, obvious cases

4. Provide Sensible Defaults

// ✅ Good - most callers won't need to specify T
interface AsyncState {
    loading: boolean;
    data?: T;
}

5. Avoid Generic Overkill

If a function only ever operates on string, don't make it generic "just in case." Generics add cognitive overhead — reserve them for genuine type-relationship needs.
